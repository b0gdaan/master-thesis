\section{Проблем}

\begin{frame}{Зашто прогнозирати зависност}
    \begin{itemize}
        \item Корелација је основа диверсификације, хеџинга и портфолио-ризика, док је анализа средње вредности и варијансе често третира као константу
        \item Она није константна. Корелације асиметрично реагују на лоше вести и расту када тржишта падају, баш када је диверсификација најпотребнија
        \item Bitcoin додатно наглашава проблем: њиме се тргује глобално и непрекидно, а његова улога на спектру ризика није фиксна
        \item Зато је предмет истраживања \textbf{променљива} величина. Једна корелација за цео узорак сакрила би важне тржишне режиме
    \end{itemize}
\end{frame}

\begin{frame}{Истраживачка питања}
    \begin{enumerate}
        \item Да ли је склизна зависност прогнозива ван узорка?
        \item Да ли машинско учење надмашује DCC-GARCH модел без цурења будућих информација?
        \item Да ли надмашује једноставне моделе постојаности, као што је AR(1)?
        \item Да ли прогнозе могу бити основа за сигнал упозорења на стрес за инвеститора?
    \end{enumerate}
\end{frame}

\section{Методологија}

\begin{frame}{\secname}
    \begin{description}
        \item[Walk-forward] проширујући прозор, $\mathrm{min\_train} = 800$, поновно оцењивање на сваких 20 корака
        \item[Модели] Naive, AR(1), HAR, Ridge, ElasticNet, RF, GBM, XGBoost и Ensemble
        \item[Референтни модел] DCC-GARCH(1,1), поново оцењен унутар петље
        \item[Тестирање] Diebold--Mariano тест на упареним разликама губитка
    \end{description}
    \bigskip
    \begin{itemize}
        \item Сваки корак који би могао открити будућност --- конструкција циља, поравнање обележја, оцена референтног модела и генерисање сигнала --- извршава се \emph{унутар} петље
        \item Хиперпараметри су једном изабрани временски унакрсном валидацијом за BTC--S\&P\,500, $w=30$; тај репрезентативни случај је зато истраживачки, а не независна потврдна оцена
    \end{itemize}
\end{frame}

\section{Закључци}

\begin{frame}{Шта рад показује}
    \begin{enumerate}
        \item Временски променљива међутржишна зависност Bitcoin-а и традиционалних актива \textbf{јесте} прогнозива ван узорка: шест парова, четири прозора и строг walk-forward
        \item Прогнозивост потиче из постојаности самог циља: један лаг зависности већ даје $R^2 = 0.885$
        \item Машинско учење има мањи RMSE од DCC-GARCH модела без цурења информација у сваком приказаном експерименту, али је поређење описно; AR(1) није доследно надмашен
        \item Слој сигнала је скроман додатни индикатор ризика, најјачи за широку изложеност акцијама, а не самостално правило темпирања
    \end{enumerate}
\end{frame}

\begin{frame}{Шта рад \emph{не} тврди}
    \begin{itemize}
        \item Bitcoin не предвиђа \emph{приносе} акција. Делимично предвиђа променљиву \emph{зависност} Bitcoin-а и акција, што је ужи и прецизнији исказ
        \item Сигнал процењује вероватноћу повећаног ризика пада; не говори о смеру, величини ни времену кретања у означеном периоду
        \item Најбоља конфигурација сигнала изабрана је међу двадесет, без корекције за вишеструка поређења, па је индикативна, а не потврђена хипотеза
        \item Рад не показује да је DCC-GARCH лош модел за оно за шта је конструисан
    \end{itemize}
\end{frame}

\begin{frame}{Ограничења и будући рад}
    \begin{itemize}
        \item Дневна учесталост и хоризонт од једног корака: ауторегресивна предност би требало да слаби са дужим хоризонтом
        \item Pearson-ова склизна корелација је једина мера зависности; мера осетљива на репове тестирала би иста питања другачије
        \item Једна основна криптовалута
        \item Нема калибрације вероватноћа у слоју сигнала ни портфолио backtest-а са трансакционим трошковима
        \item Потврдни рад треба да користи угнежђено walk-forward подешавање и HAC спецификацију која моделира преклапање склизних прозора
    \end{itemize}
    \bigskip
    {\small Свако ограничење је природан правац наставка рада. Ниједно не поништава резултате у оквиру онога што је заиста тестирано.}
\end{frame}
